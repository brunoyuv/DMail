cmake_minimum_required(VERSION 3.5.0)
project(thunderbird_native)
set(CMAKE_CXX_STANDARD 17)
set(NATIVE_LIB_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../libs/${OHOS_ARCH}")
if(NOT EXISTS "${NATIVE_LIB_DIR}/libThunderbirdCore.so")
  message(FATAL_ERROR "Run scripts/prepare-native-app before building the HAP")
endif()
add_library(ThunderbirdCore SHARED IMPORTED)
set_target_properties(ThunderbirdCore PROPERTIES IMPORTED_LOCATION "${NATIVE_LIB_DIR}/libThunderbirdCore.so")
add_library(thunderbird SHARED napi_init.cpp)
target_link_libraries(thunderbird PUBLIC ThunderbirdCore libace_napi.z.so)
target_link_options(thunderbird PRIVATE "-Wl,-rpath,$ORIGIN" "-Wl,--no-undefined")

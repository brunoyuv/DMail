cmake_minimum_required(VERSION 3.24)
project(HarmonyFTS LANGUAGES C)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "OHOS")
  message(FATAL_ERROR "This adapter requires the Huawei OHOS toolchain")
endif()
if(NOT FTS_SOURCE_DIR)
  message(FATAL_ERROR "Set FTS_SOURCE_DIR to the pinned musl-fts source")
endif()
include(CheckSymbolExists)
check_symbol_exists(MAX "sys/param.h" HAVE_DECL_MAX)
check_symbol_exists(UINTMAX_MAX "limits.h" HAVE_DECL_UINTMAX_MAX)
check_symbol_exists(dirfd "dirent.h" HAVE_DIRFD)
configure_file(config.h.in config.h)
add_library(fts SHARED "${FTS_SOURCE_DIR}/fts.c")
set_target_properties(fts PROPERTIES C_STANDARD 11 POSITION_INDEPENDENT_CODE ON)
target_include_directories(fts PRIVATE "${FTS_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}")
install(TARGETS fts LIBRARY DESTINATION lib)
install(FILES "${FTS_SOURCE_DIR}/fts.h" DESTINATION include)
